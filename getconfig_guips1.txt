# ---------------- find ttpmacro.exe ----------------
function Find-TtpMacro {
  if ($TtpMacroPath -and (Test-Path -LiteralPath $TtpMacroPath)) { return $TtpMacroPath }

  $saved = Join-Path $BaseDir 'ttpmacro_path.txt'
  if (Test-Path -LiteralPath $saved) {
    $p = (Get-Content -LiteralPath $saved -TotalCount 1)
    if ($p) { $p = $p.Trim() }
    if ($p -and (Test-Path -LiteralPath $p)) { return $p }
  }

  try {
    $progId = (Get-ItemProperty -LiteralPath 'Registry::HKEY_CLASSES_ROOT\.ttl' -ErrorAction Stop).'(default)'
    if ($progId) {
      $cmd = (Get-ItemProperty -LiteralPath "Registry::HKEY_CLASSES_ROOT\$progId\shell\open\command" -ErrorAction Stop).'(default)'
      if ($cmd -match '([A-Za-z]:\\[^"]*?ttpmacro\.exe)') {
        if (Test-Path -LiteralPath $Matches[1]) { return $Matches[1] }
      }
    }
  } catch {}

  $pf86 = [Environment]::GetEnvironmentVariable('ProgramFiles(x86)')
  foreach ($root in @($env:ProgramFiles, $pf86)) {
    if (-not $root) { continue }
    foreach ($d in @('teraterm5', 'teraterm', 'Tera Term')) {
      $c = Join-Path $root "$d\ttpmacro.exe"
      if (Test-Path -LiteralPath $c) { return $c }
    }
  }

  $dlg = New-Object System.Windows.Forms.OpenFileDialog
  $dlg.Title  = 'Select ttpmacro.exe (in Tera Term folder)'
  $dlg.Filter = 'ttpmacro.exe|ttpmacro.exe|All files|*.*'
  if ($dlg.ShowDialog() -eq [System.Windows.Forms.DialogResult]::OK) {
    Set-Content -LiteralPath $saved -Value $dlg.FileName -Encoding Default
    return $dlg.FileName
  }
  return $null
}

# ---------------- read List.txt ----------------
function Get-Field($arr, $i) {
  if ($arr.Count -gt $i) { return $arr[$i].Trim() } else { return '' }
}

function Load-List {
  $path = Join-Path $BaseDir $ListFile
  if (-not (Test-Path -LiteralPath $path)) { throw "List.txt not found:`r`n$path" }
  $items = New-Object System.Collections.ArrayList
  $no = 0
  foreach ($raw in [System.IO.File]::ReadAllLines($path, [System.Text.Encoding]::Default)) {
    $line = $raw.Trim()
    if ($line.Length -eq 0) { continue }
    if ($line -match '^[^A-Za-z0-9,]*#') { continue }   # comment line
    $no++
    $f = $line.Split(',')
    $method = (Get-Field $f 2).ToUpper()
    $port = Get-Field $f 6
    if (-not $port) { if ($method -eq 'TELNET') { $port = '23' } else { $port = '22' } }
    [void]$items.Add([pscustomobject]@{
      No = $no; HostName = (Get-Field $f 0); IP = (Get-Field $f 1)
      Method = $method; User = (Get-Field $f 3); Port = $port
      CmdFile = $(if (Get-Field $f 7) { Get-Field $f 7 } else { '(default)' })
    })
  }
  return ,$items
}

# ---------------- logic ----------------
function Update-GoButton {
  $n = 0
  foreach ($r in $grid.Rows) { if ($r.Cells['Sel'].Value -eq $true) { $n++ } }
  $lblCount.Text = "$n / $($grid.Rows.Count) selected"
  $lblCount.Left = $header.ClientSize.Width - $lblCount.Width - 26
  if ($script:running) {
    $btnGo.Text = 'STOP'
    $btnGo.Enabled = $true
    $btnGo.BackColor = $cStop
    $btnGo.FlatAppearance.MouseOverBackColor = C 190 40 55
  } else {
    $btnGo.Text = 'GO'
    $btnGo.Enabled = ($n -gt 0)
    if ($n -gt 0) { $btnGo.BackColor = $cAccent } else { $btnGo.BackColor = $cDisabled }
    $btnGo.FlatAppearance.MouseOverBackColor = $cAccentDark
  }
}

function Set-Running($on) {
  $script:running = $on
  $btnAll.Enabled = -not $on
  $btnNone.Enabled = -not $on
  $btnReload.Enabled = -not $on
  $numInt.Enabled = -not $on
  $colSel.ReadOnly = $on
  Update-GoButton
}

function Fill-Grid {
  $grid.Rows.Clear()
  $items = @()
  try { $items = Load-List } catch { Show-Error $_.Exception.Message }
  foreach ($it in $items) {
    $idx = $grid.Rows.Add($true, $it.No, $it.HostName, $it.IP, $it.Method, $it.User, $it.Port, $it.CmdFile, 'Ready')
    $row = $grid.Rows[$idx]
    $mc = $row.Cells['Method']
    $mc.Style.Font = $fontBold
    switch ($it.Method) {
      'SSH'    { $mc.Style.ForeColor = C 37 99 235 }
      'CR'     { $mc.Style.ForeColor = C 124 58 237 }
      'CRM'    { $mc.Style.ForeColor = C 192 38 211 }
      'TELNET' { $mc.Style.ForeColor = C 217 119 6 }
      default  {
        $mc.Style.ForeColor = $cStop
        $row.Cells['Sel'].Value = $false
        $row.Cells['Status'].Value = 'Invalid method'
      }
    }
    $row.Cells['Status'].Style.ForeColor = $cGray
    $row.Cells['CmdFile'].Style.ForeColor = $cGray
    if ($it.CmdFile -ne '(default)' -and -not (Test-Path -LiteralPath (Join-Path $BaseDir $it.CmdFile))) {
      $row.Cells['CmdFile'].Style.ForeColor = $cStop
      $row.Cells['Status'].Value = 'Cmd file missing'
    }
  }
  $grid.ClearSelection()
  $lblStatus.Text = "Loaded $($grid.Rows.Count) host(s) from $ListFile"
  Update-GoButton
}

$timer = New-Object System.Windows.Forms.Timer
$timer.Add_Tick({
  $timer.Stop()
  if ($script:queue.Count -eq 0) {
    Set-Running $false
    $lblStatus.Text = "Done. $($script:launched) host(s) launched.   Log: $($script:logDir)"
    return
  }
  $r = $script:queue.Dequeue()
  $no = $r.Cells['No'].Value
  try {
    Start-Process -FilePath $script:ttp -ArgumentList ('/V "{0}" {1}' -f $script:child, $no)
    $r.Cells['Status'].Value = 'Launched'
    $r.Cells['Status'].Style.ForeColor = $cAccentDark
    $r.Cells['Status'].Style.Font = $fontBold
  } catch {
    $r.Cells['Status'].Value = 'Launch error'
    $r.Cells['Status'].Style.ForeColor = $cStop
  }
  $script:launched++
  $grid.FirstDisplayedScrollingRowIndex = [Math]::Max(0, $r.Index - 3)

  $wait = [int]$numInt.Value * 1000
  $msg = "Launching...  $($script:launched) / $($script:total)"
  if ($script:queue.Count -eq 0) {
    $wait = 200
  } elseif ($BatchSize -gt 0 -and ($script:launched % $BatchSize) -eq 0) {
    $wait = $BatchWaitSec * 1000
    $msg += "   (resting $BatchWaitSec sec)"
  }
  $lblStatus.Text = $msg
  $timer.Interval = $wait
  $timer.Start()
})

$btnGo.Add_Click({
  if ($script:running) {
    # STOP: cancel hosts not yet launched
    foreach ($r in $script:queue) { $r.Cells['Status'].Value = 'Canceled' }
    $script:queue.Clear()
    $lblStatus.Text = 'Stopped. Windows already opened will continue.'
    return
  }

  $ttp = Find-TtpMacro
  if (-not $ttp) { return }
  $child = Join-Path $BaseDir $ChildMacro
  if (-not (Test-Path -LiteralPath $child)) { Show-Error "$ChildMacro not found:`r`n$child"; return }

  $targets = @()
  foreach ($r in $grid.Rows) { if ($r.Cells['Sel'].Value -eq $true) { $targets += $r } }
  if ($targets.Count -eq 0) { return }

  $ans = [System.Windows.Forms.MessageBox]::Show("Start log collection for $($targets.Count) host(s)?", 'CONFIRM', 'OKCancel', 'Question')
  if ($ans -ne [System.Windows.Forms.DialogResult]::OK) { return }

  $script:logDir = Join-Path (Join-Path $BaseDir $LogDirName) (Get-Date -Format 'yyyyMMdd')
  New-Item -ItemType Directory -Force -Path $script:logDir | Out-Null

  $script:ttp = $ttp
  $script:child = $child
  $script:queue = New-Object System.Collections.Queue
  foreach ($r in $grid.Rows) {
    if ($r.Cells['Sel'].Value -eq $true) {
      $script:queue.Enqueue($r)
      $r.Cells['Status'].Value = 'Waiting'
      $r.Cells['Status'].Style.ForeColor = $cGray
      $r.Cells['Status'].Style.Font = $fontMain
    }
  }
  $script:total = $script:queue.Count
  $script:launched = 0
  Set-Running $true
  $timer.Interval = 100
  $timer.Start()
})
